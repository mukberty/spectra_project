import torch
import numpy as np
import faiss
import torch.nn.functional as F
from torch.utils.data import DataLoader

from models.molclip_model import MolCLIP
from utils.image_retrieval_dataset import ImageRetrievalDataset
from utils.smiles_tokenizer import SmilesTokenizer


# -----------------------
# device
# -----------------------
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Using:", device)


# -----------------------
# dataset (TRAIN ONLY)
# -----------------------
dataset = ImageRetrievalDataset(
    parquet_path="data/processed/curated_subset_named.parquet",
    image_dir="data/plots/continuous_spectra"
)

loader = DataLoader(
    dataset,
    batch_size=32,
    shuffle=False,
    num_workers=2,
    pin_memory=True
)

print("Train size:", len(dataset))


# -----------------------
# SMILES + tokenizer
# -----------------------
df = dataset.df.reset_index(drop=True)

smiles_list = df["smiles"].astype(str).tolist()

tokenizer = SmilesTokenizer(smiles_list, max_len=128)


# -----------------------
# model (MUST MATCH TRAINING)
# -----------------------
ckpt = torch.load(
    "checkpoints/molclip_aligned.pt",
    map_location=device
)

model = MolCLIP(vocab_size=tokenizer.vocab_size).to(device)
model.load_state_dict(ckpt, strict=True)
model.eval()


# -----------------------
# build embeddings
# -----------------------
embeddings = []
ids = []

with torch.no_grad():

    for ir, ram, mol_id in loader:

        ir = ir.to(device)
        ram = ram.to(device)

        # IMPORTANT: SMILES must come from SAME row ordering
        batch_idx = mol_id.numpy() if torch.is_tensor(mol_id) else mol_id
        smiles_batch = df.loc[batch_idx, "smiles"].astype(str).tolist()

        smiles_batch = tokenizer.batch_encode(smiles_batch).to(device)

        # 🔥 USE IMAGE EMBEDDING (this is retrieval system)
        emb = model.encode_image(ir, ram)
        emb = F.normalize(emb, dim=1)

        embeddings.append(emb.cpu().numpy())
        ids.extend(list(mol_id))


embeddings = np.vstack(embeddings).astype("float32")
ids = np.array(ids)


# -----------------------
# FAISS INDEX
# -----------------------
index = faiss.IndexFlatIP(embeddings.shape[1])
index.add(embeddings)

faiss.write_index(index, "checkpoints/molclip.index")
np.save("checkpoints/molclip_ids.npy", ids)

print("INDEX BUILT:", index.ntotal)
